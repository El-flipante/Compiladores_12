// Vetores locais: preenchimento, soma, inversao no lugar, maximo e
// ordenacao por bolha com indices calculados.
int main() {
    int v[8] = {5, 3, 9, 1, 7, 2, 8, 4};
    int n;
    int soma;
    int maior;
    int t;
    int r;
    n = 8;
    soma = 0;
    for (int i = 0; i < n; i = i + 1) soma = soma + v[i];

    for (int i = 0; i < n / 2; i = i + 1) {
        t = v[i];
        v[i] = v[n - 1 - i];
        v[n - 1 - i] = t;
    }

    maior = v[0];
    for (int i = 1; i < n; i = i + 1) {
        if (v[i] > maior) maior = v[i];
    }

    for (int i = 0; i < n - 1; i = i + 1) {
        for (int j = 0; j < n - 1 - i; j = j + 1) {
            if (v[j] > v[j + 1]) {
                t = v[j];
                v[j] = v[j + 1];
                v[j + 1] = t;
            }
        }
    }

    r = soma * 1000 + maior * 100;
    for (int i = 0; i < n; i = i + 1) r = r + v[i] * (i + 1);
    return r;
}

// char e um byte com sinal: escritas truncam e comparacoes usam o valor truncado.
char vogal[5] = {'a', 'e', 'i', 'o', 'u'};
char palavra[8] = {'c', 'o', 'm', 'p'};

char dobra(char c) {
    return c + c;
}

char maiuscula(char c) {
    if (c >= 'a') {
        if (c <= 'z') return c - ' ';
    }
    return c;
}

int tamanho() {
    int n;
    n = 0;
    while (palavra[n] != '\0') n = n + 1;
    return n;
}

int eh_vogal(char c) {
    for (int i = 0; i < 5; i = i + 1) {
        if (vogal[i] == c) return 1;
    }
    return 0;
}

int main() {
    char c;
    char d;
    int r;
    r = 0;
    c = 'a' + 'a';
    if (c < 'a') r = r + 1;
    d = dobra('z');
    if (d < 'a') r = r + 10;
    if (dobra('A') == 'A' + 'A') r = r + 100;
    if (maiuscula('q') == 'Q') r = r + 1000;
    if (maiuscula('Q') == 'Q') r = r + 2000;
    if (maiuscula('1') == '1') r = r + 4000;
    r = r + tamanho() * 10000;
    r = r + eh_vogal('o') * 100000 + eh_vogal('x') * 200000 + eh_vogal('u') * 400000;
    return r;
}

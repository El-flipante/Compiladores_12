int potencia(int base, int exp) {
    int r;
    r = 1;
    while (exp > 0) {
        if (exp % 2 == 1) r = r * base;
        base = base * base;
        exp = exp / 2;
    }
    return r;
}

int soma_digitos(int n) {
    int s;
    s = 0;
    while (n > 0) {
        s = s + n % 10;
        n = n / 10;
    }
    return s;
}

int inverte(int n) {
    int r;
    r = 0;
    while (n > 0) {
        r = r * 10 + n % 10;
        n = n / 10;
    }
    return r;
}

int main() {
    return potencia(3, 13) + soma_digitos(987654321) * 1000 + inverte(12345);
}

// else-if encadeado e "else pendente" (liga-se ao if mais proximo).
int classifica(int n) {
    if (n < 0) return -1;
    else if (n == 0) return 0;
    else if (n < 10) return 1;
    else if (n < 100) return 2;
    else return 3;
}

int pendente(int a, int b) {
    int r;
    r = 0;
    if (a > 0)
        if (b > 0) r = 1;
        else r = 2;
    return r;
}

int aninhado(int a, int b, int c) {
    if (a > b) {
        if (b > c) return 1;
        else { if (a > c) return 2; else return 3; }
    } else {
        if (a > c) return 4;
        return 5;
    }
}

int main() {
    int r;
    r = 0;
    r = r * 5 + classifica(-9) + 1;
    r = r * 5 + classifica(0) + 1;
    r = r * 5 + classifica(7) + 1;
    r = r * 5 + classifica(42) + 1;
    r = r * 5 + classifica(1000) + 1;
    r = r * 5 + pendente(1, 1) * 9 + pendente(1, 0) * 3 + pendente(0, 5);
    r = r * 7 + aninhado(3, 2, 1) + aninhado(5, 2, 4) + aninhado(5, 1, 7) + aninhado(1, 5, 0) + aninhado(1, 5, 9);
    return r;
}

// Prototipo permite que as funcoes se chamem mutuamente.
int impar(int n);

int par(int n) {
    if (n == 0) return 1;
    return impar(n - 1);
}

int impar(int n) {
    if (n == 0) return 0;
    return par(n - 1);
}

int main() {
    return par(10) * 1000 + par(7) * 100 + impar(7) * 10 + impar(10);
}

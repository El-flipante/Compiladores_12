// Sombreamento de globais, parametros e variaveis de blocos externos.
int x;
int y;

int usa_global() {
    return x * 10 + y;
}

int sombra_parametro(int x) {
    {
        int x;
        x = 50;
        y = y + x;
    }
    return x;
}

int main() {
    int r;
    x = 1;
    y = 2;
    r = usa_global();
    {
        int x;
        x = 7;
        {
            int x;
            x = 8;
            r = r * 100 + x;
        }
        r = r * 100 + x;
        {
            int y;
            y = 9;
            r = r * 100 + y;
        }
    }
    r = r * 100 + x;
    r = r * 3 + sombra_parametro(4);
    r = r * 3 + y;
    for (int x = 0; x < 3; x = x + 1) r = r + x;
    for (int x = 10; x < 12; x = x + 1) r = r + x;
    r = r * 3 + x;
    return r;
}

// Os seis operadores relacionais, nos tres casos (<, >, ==).
int mascara(int a, int b) {
    int r;
    r = 0;
    if (a < b) r = r + 1;
    if (a > b) r = r + 2;
    if (a <= b) r = r + 4;
    if (a >= b) r = r + 8;
    if (a == b) r = r + 16;
    if (a != b) r = r + 32;
    return r;
}

int main() {
    int r;
    r = mascara(1, 2);
    r = r * 64 + mascara(2, 1);
    r = r * 64 + mascara(3, 3);
    r = r * 64 + mascara(-5, -4);
    r = r * 64 + mascara(-1, 1);
    return r;
}

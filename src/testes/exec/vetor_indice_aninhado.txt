// no mesmo comando.
int dobro(int n) { return n * 2; }

int main() {
    int a[6] = {1, 2, 3, 0, 0, 0};
    int i;
    i = 0;
    a[i] = a[i + 1] + a[a[0]];
    a[a[0] + 1] = a[3] + dobro(a[0]) + 7;
    a[dobro(1) + 1] = a[a[1]] - a[i];
    return a[0] * 100000 + a[1] * 10000 + a[2] * 1000 + a[3] * 100 + a[4] * 10 + a[5];
}

// while, for com declaracao, for com atribuicao, for com partes vazias,
// lacos aninhados e escopo da variavel do for reutilizado.
int main() {
    int soma;
    int i;
    int n;
    soma = 0;
    n = 100;
    i = 1;
    while (i <= n) {
        soma = soma + i;
        i = i + 1;
    }
    for (int j = 0; j < 10; j = j + 1) {
        soma = soma + j * j;
    }
    for (int j = 10; j > 0; j = j - 3) {
        soma = soma - j;
    }
    for (i = 0; i < 5; i = i + 1) {
        for (n = 0; n < i; n = n + 1) {
            soma = soma + i * n;
        }
    }
    i = 0;
    for (; i < 7; ) {
        i = i + 2;
    }
    soma = soma * 10 + i;
    for (;;) {
        soma = soma + 1;
        if (soma > 5000) return soma;
    }
    return 0;
}

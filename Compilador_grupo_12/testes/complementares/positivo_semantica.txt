int percorrer() {
    int valores[4] = {1, 2, 3, 4};
    int acumulado;
    acumulado = 0;
    for (int i = 0; i < 4; i = i + 1) {
        acumulado = acumulado + valores[i];
    }
    while (acumulado > 0) {
        acumulado = acumulado - 1;
    }
    valores[0] = -5 % 2;
    acumulado = valores[0];
    return acumulado;
}

int comparacoes(int n) {
    if (n <= 10) {
        if (n >= 0) {
            if (n == 5) return 1;
            else if (n != 3) return 2;
            else return 3;

    } else return 5;
}
